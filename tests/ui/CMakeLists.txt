cmake_minimum_required(VERSION 3.21)
project(KMusicInputTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickTest)
qt_standard_project_setup(REQUIRES 6.8)
get_filename_component(KMUSIC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${KMUSIC_ROOT}/vendor/nextkdeAppDemo/shared/qml" kos-ui)
qt_add_executable(kmusic_input_tests main.cpp)
set_source_files_properties("${KMUSIC_ROOT}/qml/MusicColors.qml" PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
set(INPUT_QML MusicTextField.qml MusicActionButton.qml MusicColors.qml UiLanguage.mjs SearchDialog.qml MusicNavigationButton.qml SearchHistory.mjs)
set(INPUT_SOURCES)
foreach(name IN LISTS INPUT_QML)
    set_source_files_properties("${KMUSIC_ROOT}/qml/${name}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
    list(APPEND INPUT_SOURCES "${KMUSIC_ROOT}/qml/${name}")
endforeach()
qt_add_qml_module(kmusic_input_tests URI NextKde.Music VERSION 1.0 RESOURCE_PREFIX /qt/qml
    QML_FILES ${INPUT_SOURCES})
target_link_libraries(kmusic_input_tests PRIVATE Qt6::QuickTest Kos::Ui Kos::UiPlugin)
