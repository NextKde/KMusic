project(KMusicProviderTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Test)
add_subdirectory(../../src/providers provider-core)
add_executable(provider_contract_tests provider_contract_tests.cpp)
target_link_libraries(provider_contract_tests PRIVATE kmusic_provider_core Qt6::Test)
enable_testing()
add_test(NAME provider_contract_tests COMMAND provider_contract_tests)
add_executable(bilibili_flow_tests bilibili_flow_tests.cpp)
target_link_libraries(bilibili_flow_tests PRIVATE kmusic_provider_core Qt6::Test)
add_test(NAME bilibili_flow_tests COMMAND bilibili_flow_tests)
option(KMUSIC_TEST_PLAYBACK_HEADERS "Test native HTTP audio and header isolation" OFF)
if(KMUSIC_TEST_PLAYBACK_HEADERS)
    find_package(PkgConfig REQUIRED)

    target_link_libraries(playback_headers_tests PRIVATE Qt6::Core Qt6::Network Qt6::Test PkgConfig::GST)
    add_test(NAME playback_headers_tests COMMAND playback_headers_tests)
